新版不需要生成文件：

```sh
# 随机打开一个不包含.txt或.bc!文件的目录
flist=$(ls -d */|cut -d/ -f1|sort -u)
flist=$(echo "$flist"; ls */*.txt|cut -d/ -f1|sort -u)
flist=$(echo "$flist"; ls */*.bc\!|cut -d/ -f1|sort -u)
explorer "$(echo "$flist"|sort|uniq -c|grep -E ^\\s+1|cut -d' ' -f8-|shuf -n1)"
```

旧版：

```sh
# 随机打开一个目录
explorer `ls -d */|cut -d/ -f1|sort -u|shuf -n1`

# 随机打开一个不包含.txt或.bc!文件的目录
ls -d */|cut -d/ -f1|sort -u>tmp.tmp
ls */*.txt|cut -d/ -f1|sort -u>>tmp.tmp
ls */*.bc\!|cut -d/ -f1|sort -u>>tmp.tmp
explorer "$(cat tmp.tmp|sort|uniq -c|grep -E ^\\s+1|cut -d' ' -f8-|shuf -n1)"
```
