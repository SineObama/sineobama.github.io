## 让浏览器导出的收藏夹文件链接显示收藏日期

```js
// 在浏览器打开导出的收藏夹html文件，在其中运行，把中文日期打印出来，分离收藏名称和网址
function showFavorDate() {
    var s = document.getElementsByTagName('a');
    for (var i = 0; i < s.length; i++) {
        var el = s[i];
        var span = document.createElement('span');
        var time = el.getAttribute('ADD_DATE');
        var date = new Date(time * 1000);
        span.innerText = date.getFullYear() + '年' + (date.getMonth() + 1) + '月' + date.getDate() + '日\t';
        el.parentElement.insertBefore(span, el);
      
        var span = document.createElement('span');
        span.innerText = el.innerText + '\t';
        el.parentElement.insertBefore(span, el);
      
        el.innerText = el.getAttribute('href');
    }
}
showFavorDate()
```
