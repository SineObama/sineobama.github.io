## 背景

> Windows系统的实际使用中，如果一个文件夹包含数万的子文件夹与数万的文件，全部删除的操作会比较慢（例如会持续几分钟），这可能跟操作系统的操作方式有关。是否可以通过代码程序进行删除，减少消耗的时间？请以nodejs环境为例给出示例代码

### 智谱清言第一版代码

智谱清言给出的代码，我加了日志打印。每秒大概几千个文件夹。

```js
const fs = require('fs');
const path = require('path');

var c = 0;

function addLog() {
    c++;
    if (c % 1000 === 0) {
        console.log(c);
    }
}

// 递归删除目录及其所有内容的函数
function deleteDirectory(directoryPath) {
    if (fs.existsSync(directoryPath)) {
        fs.readdirSync(directoryPath).forEach((file, index) => {
            const curPath = path.join(directoryPath, file);
            if (fs.lstatSync(curPath).isDirectory()) { // 如果是文件夹，递归调用
                deleteDirectory(curPath);
            } else { // 如果是文件，直接删除
                fs.unlinkSync(curPath);
                addLog();
            }
        });
        fs.rmdirSync(directoryPath); // 删除空目录
        addLog();
    }
}

// 使用函数
const directoryPath = 'D:\\MyAppData\\Tencent Files\\714186139\\Image\\Group2'; // 要删除的目录路径
deleteDirectory(directoryPath);
```

### 智谱清言queue版本

这个版本还需要安装async库，我还没有运行过。

```js
const fs = require('fs').promises;
const path = require('path');
const async = require('async');

// 递归删除目录及其所有内容的函数
async function deleteDirectoryWithQueue(directoryPath, concurrency) {
    try {
        // 创建一个队列，限制并发执行的任务数量
        const q = async.queue(async (taskPath, callback) => {
            try {
                const stats = await fs.lstat(taskPath);
                if (stats.isDirectory()) {
                    // 如果是目录，递归删除
                    await deleteDirectoryWithQueue(taskPath, concurrency);
                } else {
                    // 如果是文件，直接删除
                    await fs.unlink(taskPath);
                }
                callback();
            } catch (error) {
                callback(error);
            }
        }, concurrency);

        // 处理完成后，删除空目录
        q.drain = async () => {
            try {
                await fs.rmdir(directoryPath);
                console.log(`Deleted directory: ${directoryPath}`);
            } catch (error) {
                console.error(`Failed to delete directory: ${directoryPath}`, error);
            }
        };

        // 读取目录内容
        const files = await fs.readdir(directoryPath);

        // 将每个文件/目录添加到队列中
        files.forEach((file) => {
            q.push(path.join(directoryPath, file));
        });
    } catch (error) {
        // 处理错误
        console.error(`Error deleting directory: ${directoryPath}`, error);
    }
}

// 使用函数
const directoryPath = 'path/to/directory'; // 要删除的目录路径
const concurrency = 10; // 并发删除的数量限制
deleteDirectoryWithQueue(directoryPath, concurrency);
```

### 互联网Promise版本

> 我看过另一个使用Promise实现的删除文件方法，它在面对大数据量文件时，启动会很慢（很久才开始真正删除文件夹），我猜测是因为它在执行时会先完成构建所有文件夹的Promise对象，导致启动很慢。如果像你说的这样使用这个队列方式，是否会存在相似问题？

此方法与智谱清言queue版本代码的对比，它回答确实如此。

（我加了日志打印）

```js
// import { unlink, rmdir, readdir, stat } from 'node:fs/promises'
// import { join } from 'node:path'
const {unlink, rmdir, readdir, stat} = require('fs').promises
const {join} = require('path')

var c = 0;

function addLog() {
    c++;
    if (c % 1000 === 0) {
        console.log(c);
    }
}

const removeFile = (path) => {
    return new Promise((resolve, reject) => {
        checkStat(path)
            .then(isFile => {
                if (isFile) {
                    return unlink(path)
                        .then(addLog)
                        .then(resolve, reject)
                }
                return readdir(path).then(files => {
                    const removeFiles = files.map(file => removeFile(join(path, file)))
                    return Promise.all(removeFiles)
                }).then((files) => {
                    rmdir(path)
                        .then(addLog)
                        .then(resolve, reject)
                })
            })

    })
}

const checkStat = (path) => {
    return new Promise((resolve, reject) => {
        stat(path)
            .then(stat => {
                resolve(stat.isFile())
            })
            .catch(err => {
                reject(err)
            })
    })
}

removeFile('D:\\MyAppData\\Tencent Files\\714186139\\Image\\Group2')
```
